public class App {

    public static void main(String[] args) {

        System.out.println("[System] Inizializzazione del Motore ECS in corso...");

package controller.core;

import model.data.Config;
import model.event.EventManager;
import model.phase.PhaseManager;
import model.scene.Scene;
import model.service.TimeManager;

/**
 * Gestore del ciclo di vita principale dell'applicazione (Main Loop).
 * Implementa Runnable per l'esecuzione asincrona su un Thread dedicato.
 * Calcola il Delta Time, aggiorna le metriche globali tramite TimeManager,
 * orchestra il motore logico (Engine), propaga gli eventi e aggiorna la grafica (Panel).
 */
public class Flow implements Runnable {

    private Thread thread;
    private final PhaseManager phaseManager;
    private final TimeManager timeManager;
    private boolean running;

    public Flow(PhaseManager phaseManager) {
        this.phaseManager = phaseManager;
        this.timeManager  = TimeManager.getInstance();
        this.running      = false;
    }

    public void startLoop() {
        if (this.thread == null) {
            this.running = true;
            this.thread  = new Thread(this);
            this.thread.start();
        }
    }

    public void stopLoop() {
        this.running = false;
    }

    @Override
    public void run() {
        long lastTime = System.nanoTime();
        long currentTime, elapsedTime;
        long timer = System.currentTimeMillis();
        int  frames = 0;

        while (this.running) {
            currentTime  = System.nanoTime();
            elapsedTime  = currentTime - lastTime;

            if (elapsedTime >= Config.TARGET_TIME_NS) {

                double deltaTime = elapsedTime / 1_000_000_000.0;

                // 1. Aggiornamento dati temporali
                this.timeManager.updateTimeData(deltaTime, frames);

                // 2. Battito Logico: l'Engine aggiorna la logica della scena
                this.phaseManager.getEngine().update();

                // 3. Smistamento Eventi
                EventManager.getInstance().processPendingEvents();

                // 4. Ponte Model → View
                Scene activeScene = this.phaseManager.getEngine().getActiveScene();
                this.phaseManager.getPanel().setEntities(
                    activeScene != null ? activeScene.getEntities() : null
                );

                // 5. Battito Grafico
                this.phaseManager.getPanel().repaint();

                lastTime = currentTime;
                frames++;

            } else {
                try { Thread.sleep(1); }
                catch (InterruptedException e) { Thread.currentThread().interrupt(); }
            }

            if (System.currentTimeMillis() - timer > 1000) {
                timer += 1000;
                frames = 0;
            }
        }
    }
}

package controller.input;

import java.awt.event.KeyEvent;
import java.awt.event.KeyListener;
import java.util.Observable;

/**
 * Singleton KeyListener.
 * Mappa lo stato fisico dei tasti in un array booleano[256] e notifica
 * gli Observer registrati ad ogni cambio di stato.
 */
@SuppressWarnings("deprecation")
public class EventManager extends Observable implements KeyListener {

    private static EventManager instance;
    private final boolean[] keys;

    private EventManager() {
        this.keys = new boolean[256];
    }

    public static EventManager getInstance() {
        if (instance == null) instance = new EventManager();
        return instance;
    }

    public boolean isKeyPressed(int keyCode) {
        if (keyCode < 0 || keyCode >= this.keys.length) return false;
        return this.keys[keyCode];
    }

    @Override
    public void keyPressed(KeyEvent e) {
        int code = e.getKeyCode();
        if (code >= 0 && code < this.keys.length) {
            this.keys[code] = true;
            setChanged();
            notifyObservers(e);
        }
    }

    @Override
    public void keyReleased(KeyEvent e) {
        int code = e.getKeyCode();
        if (code >= 0 && code < this.keys.length) {
            this.keys[code] = false;
            setChanged();
            notifyObservers(e);
        }
    }

    @Override
    public void keyTyped(KeyEvent e) { }

    public void resetKeys() {
        for (int i = 0; i < this.keys.length; i++) this.keys[i] = false;
    }
}

package model.component.logic.behavior.process;

import model.component.Component;
import model.component.logic.Updatable;
import model.entity.Entity;
import model.event.EventManager;
import model.event.EventMessage;
import model.event.EventType;
import model.service.TimeManager;

/**
 * Conto alla rovescia invisibile.
 * Accumula deltaTime e, raggiunta la soglia, spedisce un EventMessage
 * sull'Event Bus e si auto-disattiva.
 */
public class TimerEventBehavior implements Component, Updatable {

    private final Entity sender;
    private final double targetTimeInSeconds;
    private final EventType eventType;
    private double accumulatedTime;
    private boolean active;

    public TimerEventBehavior(Entity sender, long durationMs, EventType eventType) {
        this.sender              = sender;
        this.targetTimeInSeconds = durationMs / 1000.0;
        this.eventType           = eventType;
        this.accumulatedTime     = 0.0;
        this.active              = true;
    }

    @Override
    public void update() {
        if (this.active) {
            double dt = TimeManager.getInstance().getDeltaTime();
            this.accumulatedTime += dt;

            if (this.accumulatedTime >= this.targetTimeInSeconds) {
                this.active = false;
                System.out.println("[TimerEventBehavior] Tempo scaduto! Spedisco: " + this.eventType);
                EventManager.getInstance().enqueue(new EventMessage(this.eventType, this.sender));
            }
        }
    }
}

package model.component.logic.primitive.math;

import model.component.logic.primitive.Primitive;
import model.service.TimeManager;

/**
 * Primitivo generatore di segnale numerico ciclico.
 * Cresce nel tempo con deltaTime e si azzera raggiunto il limite.
 */
public class Counter implements Primitive {

    private final double limit;
    private final double speed;
    private double value;

    public Counter(double limit, double speed, double start) {
        this.limit = limit;
        this.speed = speed;
        this.value = start;
    }

    @Override
    public void update() {
        double dt  = TimeManager.getInstance().getDeltaTime();
        this.value = (this.value + (this.speed * dt)) % this.limit;
    }

    public double getValue() { return this.value; }
}

 * Contenitore universale ECS.

 */
public class Entity {

    private final Map<Class<? extends Component>, Component> components;

    public Entity(String id) {
        this.components = new HashMap<>();
        this.addComponent(new IdentityComponent(id));
    }

    public void addComponent(Component component) {
        this.components.put(component.getClass(), component);
    }

    public <T extends Component> T getComponent(Class<T> componentClass) {
        return componentClass.cast(this.components.get(componentClass));
    }

    public Collection<Component> getComponents() {
        return this.components.values();
    }

    public String getId() {
        return this.getComponent(IdentityComponent.class).getId();
    }
}

package model.scene;

import model.component.Component;
import model.component.logic.Updatable;
import model.entity.Entity;
import java.util.*;

/**
 * Contenitore dinamico della scena attiva.
 * Mantiene una Map delle Entity per ID e una lista cachedUpdatables
 * per evitare scansioni ad ogni frame.
 */
public class Scene {

    private final Map<String, Entity> entityMap;
    private final List<Updatable>     cachedUpdatables;

    public Scene() {
        this.entityMap        = new LinkedHashMap<>();
        this.cachedUpdatables = new ArrayList<>();
    }

    public void addEntity(Entity entity) {
        this.entityMap.put(entity.getId(), entity);
        for (Component c : entity.getComponents())
            if (c instanceof Updatable) this.cachedUpdatables.add((Updatable) c);
    }

    public void removeEntity(String id) {
        Entity e = this.entityMap.remove(id);
        if (e != null)
            for (Component c : e.getComponents())
                if (c instanceof Updatable) this.cachedUpdatables.remove((Updatable) c);
    }

    public void clearScene() {
        this.entityMap.clear();
        this.cachedUpdatables.clear();
    }

    public Iterable<Entity> getEntities()      { return this.entityMap.values(); }
    public List<Updatable>  getUpdatables()    { return this.cachedUpdatables; }
}

package model.event;

import java.util.LinkedList;
import java.util.Observable;
import java.util.Queue;

/**
 * Singleton Observable — Event Bus globale.
 * Accoda EventMessage con enqueue() e li processa con processPendingEvents()
 * (chiamato dal Flow ogni frame), notificando in broadcast tutti gli Observer.
 */
@SuppressWarnings("deprecation")
public class EventManager extends Observable {

    private static EventManager instance;
    private final Queue<EventMessage> eventQueue;

    private EventManager() { this.eventQueue = new LinkedList<>(); }

    public static EventManager getInstance() {
        if (instance == null) instance = new EventManager();
        return instance;
    }

    public void enqueue(EventMessage message) {
        this.eventQueue.offer(message);
    }

    public void processPendingEvents() {
        while (!this.eventQueue.isEmpty()) {
            EventMessage msg = this.eventQueue.poll();
            super.setChanged();
            super.notifyObservers(msg);
        }
    }
}

 */
@SuppressWarnings("deprecation")
public class MenuPhase implements Phase, Observer {

    private final PhaseManager phaseManager;
    private final StateManager stateManager;
    private boolean isFirstLaunch = true;

    public MenuPhase(PhaseManager phaseManager) {
        this.phaseManager = phaseManager;
        this.stateManager = new StateManager(phaseManager.getSceneLoader(), phaseManager.getEngine());

        if (this.isFirstLaunch) {
            this.phaseManager.setStatus(GameStatus.CINEMATIC_PROLOGUE);
            this.stateManager.changeState(new CinematicPrologueState(this.stateManager));
            this.isFirstLaunch = false;
        } else {
            this.phaseManager.setStatus(GameStatus.MENU);
            this.stateManager.changeState(new MainMenuState(this.stateManager));
        }
    }

                    && this.phaseManager.getCurrentStatus() == GameStatus.CINEMATIC_PROLOGUE) {

                this.phaseManager.setStatus(GameStatus.MENU);
            }
        }
    }

    @Override
    public void onExit() {
        this.stateManager.changeState(null);
        EventManager.getInstance().deleteObserver(this);

    }

     * Entità immagine scalata con effetto "cover" (riempie l'intera area target).
     */
    public static Entity createAdaptiveImage(String entityId, String imagePath,
                                              int targetWidth, int targetHeight,
                                              RenderLayer layer) {
        BufferedImage image = ImageManager.getInstance().getImage(imagePath);
        double finalScale   = 1.0;
        if (image != null) {
            double scaleX = (double) targetWidth  / image.getWidth();
            double scaleY = (double) targetHeight / image.getHeight();
            finalScale    = Math.max(scaleX, scaleY);

        Entity entity    = new Entity(entityId);
        ImageReferenceComponent imageRef = new ImageReferenceComponent(imagePath);
        entity.addComponent(new Transform(0, 0, 0, finalScale));
        entity.addComponent(imageRef);
        entity.addComponent(new SpriteComponent(imageRef));
        entity.addComponent(new RenderLayerComponent(layer));
        return entity;
    }

            }

    }

package view.component;

import model.component.spatial.Transform;
import model.component.visual.media.sprite.SpriteComponent;
import model.entity.Entity;
import view.service.ImageManager;
import java.awt.*;
import java.awt.geom.AffineTransform;
import java.awt.image.BufferedImage;

/**
 * Utility stateless per il rendering di una singola Entity ECS.
 * Legge Transform e SpriteComponent e disegna su Graphics2D
 * applicando scala, rotazione e opacità.
 */
public class SpriteRenderer {

    public static void render(Entity entity, Graphics2D g2d) {
        SpriteComponent spriteData = entity.getComponent(SpriteComponent.class);
        Transform transform        = entity.getComponent(Transform.class);

        if (spriteData == null || transform == null || !spriteData.isVisible()) return;

        BufferedImage image = ImageManager.getInstance().getImage(
            spriteData.getImageReference().getImageId()
        );
        if (image == null) return;

        AffineTransform oldTransform = g2d.getTransform();
        Composite       oldComposite = g2d.getComposite();

        int    x        = (int) transform.getX();
        int    y        = (int) transform.getY();
        double rotation = transform.getRotation();
        double scale    = transform.getScale();
        float  opacity  = spriteData.getOpacity();

        if (opacity < 1.0f)
            g2d.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_OVER, opacity));

        int drawWidth  = (int) (image.getWidth()  * scale);
        int drawHeight = (int) (image.getHeight() * scale);

        if (rotation != 0.0)
            g2d.rotate(Math.toRadians(rotation), x + drawWidth / 2.0, y + drawHeight / 2.0);

        g2d.drawImage(image, x, y, drawWidth, drawHeight, null);

        g2d.setTransform(oldTransform);
        g2d.setComposite(oldComposite);
    }
}

/**

 */
public class ImageManager {

    private static ImageManager instance;
    private final Map<String, BufferedImage> images;

    private ImageManager() { this.images = new HashMap<>(); }

    public static ImageManager getInstance() {
        if (instance == null) instance = new ImageManager();
        return instance;
    }

    public void load(String path) {
        if (this.images.containsKey(path)) return;
        try {
            this.images.put(path, ImageIO.read(new File(path)));
        } catch (IOException e) {
            System.err.println("[ImageManager] Errore caricamento: " + path);
        }
    }

    public BufferedImage getImage(String path)    { return this.images.get(path); }

    public BufferedImage getSubImage(String path, int x, int y, int width, int height) {
        BufferedImage sheet = this.images.get(path);
        return (sheet != null) ? sheet.getSubimage(x, y, width, height) : null;
    }

    public void clear() { this.images.clear(); }
}

package view.service;

import model.component.data.render.RenderLayerComponent;
import model.component.visual.media.sprite.SpriteComponent;
import model.entity.Entity;
import view.component.SpriteRenderer;
import java.awt.*;
import java.awt.geom.AffineTransform;
import java.util.*;
import java.util.List;

/**
 * Sistema ECS di rendering.
 * Filtra le entità visibili, le ordina per RenderLayer (algoritmo del pittore),
 * applica l'offset telecamera per gli elementi di mondo e delega il disegno
 * a SpriteRenderer.
 */
public class RenderSystem {

    private final Comparator<Entity> layerComparator;

    public RenderSystem() {
        this.layerComparator = Comparator.comparingInt(e ->
            e.getComponent(RenderLayerComponent.class).getLayer().ordinal()
        );
    }

    public void render(Graphics2D g2d, Iterable<Entity> entities,
                       double cameraX, double cameraY) {
        List<Entity> renderables = new ArrayList<>();

        // 1. Filtraggio
        for (Entity e : entities) {
            SpriteComponent       sprite = e.getComponent(SpriteComponent.class);
            RenderLayerComponent  layer  = e.getComponent(RenderLayerComponent.class);
            if (sprite != null && layer != null && sprite.isVisible())
                renderables.add(e);
        }

        // 2. Ordinamento Z-Index
        renderables.sort(this.layerComparator);

        // 3. Ciclo di disegno
        for (Entity element : renderables) {
            AffineTransform saved  = g2d.getTransform();
            boolean isUI           = element.getComponent(RenderLayerComponent.class)
                                            .getLayer().name().startsWith("UI");
            if (!isUI) g2d.translate(-cameraX, -cameraY);
            SpriteRenderer.render(element, g2d);
            g2d.setTransform(saved);
        }
    }
}
